cmake_minimum_required(VERSION 3.20)
project(CommonwealthOnlineGnsClientBridge LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
endif()

set(_CO_BUNDLED_GNS_SOURCE_DIR "${CMAKE_CURRENT_LIST_DIR}/../../ThirdParty/GameNetworkingSockets-src")
set(GNS_SOURCE_DIR "${_CO_BUNDLED_GNS_SOURCE_DIR}" CACHE PATH "Pinned GameNetworkingSockets source tree")
if(NOT EXISTS "${GNS_SOURCE_DIR}/CMakeLists.txt" AND EXISTS "${_CO_BUNDLED_GNS_SOURCE_DIR}/CMakeLists.txt")
    message(WARNING "GNS_SOURCE_DIR does not contain an initialized GameNetworkingSockets tree: ${GNS_SOURCE_DIR}; using bundled pinned source: ${_CO_BUNDLED_GNS_SOURCE_DIR}")
    set(GNS_SOURCE_DIR "${_CO_BUNDLED_GNS_SOURCE_DIR}" CACHE PATH "Pinned GameNetworkingSockets source tree" FORCE)
endif()
if(NOT EXISTS "${GNS_SOURCE_DIR}/CMakeLists.txt")
    message(FATAL_ERROR "GNS_SOURCE_DIR must point to the initialized pinned GameNetworkingSockets source tree.")
endif()

set(BUILD_STATIC_LIB ON CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIB OFF CACHE BOOL "" FORCE)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(ENABLE_ICE OFF CACHE BOOL "" FORCE)
set(USE_STEAMWEBRTC OFF CACHE BOOL "" FORCE)
if(WIN32)
    set(USE_CRYPTO BCrypt CACHE STRING "" FORCE)
    set(Protobuf_USE_STATIC_LIBS ON CACHE BOOL "" FORCE)
else()
    set(USE_CRYPTO OpenSSL CACHE STRING "" FORCE)
    set(USE_CRYPTO25519 OpenSSL CACHE STRING "" FORCE)
endif()

set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
    set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/bin")
    set(CMAKE_LIBRARY_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/bin")
endforeach()

add_subdirectory("${GNS_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/gns")

add_library(commonwealth_online_gns_bridge SHARED
    co_gns_client_bridge.cpp
)
target_include_directories(commonwealth_online_gns_bridge PUBLIC "${CMAKE_CURRENT_LIST_DIR}")
target_compile_definitions(commonwealth_online_gns_bridge PRIVATE CO_GNS_BRIDGE_BUILD)
if(MSVC)
    target_compile_options(commonwealth_online_gns_bridge PRIVATE /W4 /WX /permissive-)
else()
    target_compile_options(commonwealth_online_gns_bridge PRIVATE -Wall -Wextra -Werror)
endif()
target_link_libraries(commonwealth_online_gns_bridge PRIVATE GameNetworkingSockets::static)
set_target_properties(commonwealth_online_gns_bridge PROPERTIES OUTPUT_NAME "commonwealth_online_gns_bridge")

option(CO_GNS_BUILD_TESTS "Build client bridge loader/export smoke test" ON)
if(CO_GNS_BUILD_TESTS)
    include(CTest)
    add_executable(co_gns_client_bridge_exports_test test_exports.cpp)
    if(MSVC)
        target_compile_options(co_gns_client_bridge_exports_test PRIVATE /W4 /WX /permissive-)
    else()
        target_compile_options(co_gns_client_bridge_exports_test PRIVATE -Wall -Wextra -Werror)
    endif()
    target_link_libraries(co_gns_client_bridge_exports_test PRIVATE commonwealth_online_gns_bridge)
    add_test(NAME co-gns-client-bridge-exports COMMAND co_gns_client_bridge_exports_test)
endif()
